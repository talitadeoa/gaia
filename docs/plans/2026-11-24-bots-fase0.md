# Bots especialistas — Fase 0 (fundação) — Implementation Plan

> **For:** quem for executar este plano. Use `superpowers:executing-plans`
> task-by-task. Design de referência: `2026-11-24-bots-especialistas-design.md`.
> Referência atemporal do repo: `../../AGENTS.md`.

**Goal:** deixar o motor pronto para vários bots ao mesmo tempo — eventos por
sessão, política de ferramentas/workspace por sessão, schema v12 e índice com
extensão configurável. Nenhuma UI nova, nenhum bot no ar ainda.

**Architecture:** quatro mudanças independentes, cada uma com teste próprio. A
ordem importa: (1) eventos por sessão é pré-requisito de qualquer execução
concorrente; (2) política por sessão é pré-requisito do isolamento; (3) schema
v12 é só banco; (4) `text_extensions` é só o índice. Nada disso depende de TOML
de bot, então dá para revisar cada peça isolada.

**Tech Stack:** Python 3.12, asyncio, SQLite (WAL), pydantic, pytest.

## Contexto para quem nunca viu este repo

- O motor é `AgentClient` (`gaia/agent/runtime.py`). Ele guarda
  `self._events: asyncio.Queue[Event]` — **uma fila global** — e
  `self._sessions: dict[str, AgentSession]`.
- `Orchestrator` (`gaia/controller/turns.py`) consome essa fila com
  `async for event in self.client.stream_events(stop=stop)` e **descarta
  eventos de outras sessões** (`if getattr(event, "session_id","") == sid`).
  Isso esconde o problema hoje: com um turno por vez, a fila global basta.
- `prompt_async` chama `_drain_events()` antes de começar. Com dois turnos
  simultâneos, o segundo **descarta os eventos do primeiro** (que fica mudo até
  o `stall_timeout`). É o bug que esta fase corrige.
- `ToolContext` (`gaia/agent/tools/base.py`) é criado **uma vez** no
  `AgentClient.__init__` e reusado por todas as sessões. Não há como dar
  `directory`/`scope`/`tools` diferentes por bot hoje.
- Verificação obrigatória antes de cada commit:
  `.venv/bin/python -m pytest -q && .venv/bin/ruff check gaia tests && .venv/bin/mypy`
- Comentários e mensagens em português sem acento ("sessao", "execucao").

```python
@dataclass(frozen=True)
class SessionPolicy:
    """Restricao de capacidade de uma sessao (bot).

    ``tools=None`` significa "todas as registradas" — e o comportamento de
    antes dos bots, preservado. ``directory`` e o cwd das ferramentas (bash,
    pi); ``scope`` e o teto de leitura/escrita do filesystem. Os dois podem ser
    diferentes: Donna escreve em ``~/dissertacao`` mas le em ``~/SecondBrain``.
    """

    tools: tuple[str, ...] | None = None
    directory: str = ""
    scope: str = ""
    bot: str = ""

    def to_dict(self) -> dict[str, Any]:
        return {
            "tools": list(self.tools) if self.tools is not None else None,
            "directory": self.directory,
            "scope": self.scope,
            "bot": self.bot,
        }

    @classmethod
    def from_dict(cls, data: dict[str, Any]) -> SessionPolicy:
        tools = data.get("tools")
        return cls(
            tools=tuple(tools) if isinstance(tools, (list, tuple)) else None,
            directory=str(data.get("directory") or ""),
            scope=str(data.get("scope") or ""),
            bot=str(data.get("bot") or ""),
        )
```

```python
    """Le a fila da sessao ate o turno terminar e devolve o texto emitido."""

    from gaia.agent.events import TurnCompleted, TurnFailed

    partes: list[str] = []
    while True:
        evento = await asyncio.wait_for(fila.get(), timeout=timeout)
        if isinstance(evento, EventTextDelta):
            partes.append(evento.text)
        elif isinstance(evento, (TurnCompleted, TurnFailed)):
            return "".join(partes)
```

Substitua `_emit` e `_drain_events`:

```python
    def _emit(self, event: Event) -> None:
        self._events.put_nowait(event)
        # Fan-out: cada assinante da sessao recebe a propria copia. Sem isso,
        # dois turnos simultaneos disputam a fila global e um deles fica mudo.
        for fila in list(self._session_queues.get(event.session_id, ())):
            fila.put_nowait(event)

    def subscribe_session(self, session_id: str) -> asyncio.Queue[Event]:
        """Fila so desta sessao. Quem assina precisa desassinar no `finally`."""
        fila: asyncio.Queue[Event] = asyncio.Queue()
        self._session_queues.setdefault(session_id, []).append(fila)
        return fila

    def unsubscribe_session(
        self, session_id: str, fila: asyncio.Queue[Event]
    ) -> None:
        filas = self._session_queues.get(session_id)
        if not filas:
            return
        if fila in filas:
            filas.remove(fila)
        if not filas:
            self._session_queues.pop(session_id, None)

    def _drain_events(self, session_id: str = "") -> None:
        """Descarta eventos pendentes de um turno anterior interrompido.

        Com ``session_id`` limpa so as filas daquela sessao. A fila global so e
        limpa quando o chamador nao pede sessao especifica (caminho antigo):
        limpar a global num turno concorrente apagaria eventos de outro.
        """
        if session_id:
            for fila in list(self._session_queues.get(session_id, ())):
                self._descartar(fila)
            return
        self._descartar(self._events)

    @staticmethod
    def _descartar(fila: asyncio.Queue[Event]) -> None:
        while True:
            try:
                fila.get_nowait()
            except asyncio.QueueEmpty:
                return
```

```python
        self, session_id: str, stop: asyncio.Event | None = None
```

```python
    def add_bot_message(
        self, bot: str, kind: str = "job", title: str = "", text: str = ""
    ) -> int:
        """Grava um item do feed do bot (resultado de job ou recado entre bots)."""
        with self._lock, self._conn:
            cursor = self._conn.execute(
                """
                INSERT INTO bot_messages (bot, kind, title, text, seen, created_at)
                VALUES (?, ?, ?, ?, 0, ?)
                """,
                (bot, kind, title, text, _now()),
            )
            return int(cursor.lastrowid or 0)

    def list_bot_messages(
        self, bot: str, limit: int = 50, unread_only: bool = False
    ) -> list[dict[str, Any]]:
        """Feed do bot, mais novo primeiro."""
        sql = "SELECT * FROM bot_messages WHERE bot = ?"
        valores: list[Any] = [bot]
        if unread_only:
            sql += " AND seen = 0"
        sql += " ORDER BY created_at DESC, id DESC LIMIT ?"
        valores.append(int(limit))
        with self._lock:
            rows = self._conn.execute(sql, valores).fetchall()
        return [dict(row) for row in rows]

    def mark_bot_seen(self, bot: str) -> int:
        """Marca tudo como lido e devolve quantos itens mudaram."""
        with self._lock, self._conn:
            cursor = self._conn.execute(
                "UPDATE bot_messages SET seen = 1 WHERE bot = ? AND seen = 0",
                (bot,),
            )
            return cursor.rowcount

    def unread_by_bot(self) -> dict[str, int]:
        """Contagem de nao-lidos por bot (o badge do rail)."""
        with self._lock:
            rows = self._conn.execute(
                "SELECT bot, COUNT(*) AS total FROM bot_messages "
                "WHERE seen = 0 GROUP BY bot"
            ).fetchall()
        return {row["bot"]: int(row["total"]) for row in rows}
```

```python
        if not self.root.exists():
            return
        excluded = {name.casefold() for name in self.cfg.exclude}
        extensoes = {
            ext if ext.startswith(".") else f".{ext}"
            for ext in (self.cfg.text_extensions or [".md"])
        }
        for dirpath, dirnames, filenames in os.walk(self.root):
            dirnames[:] = [
                d for d in dirnames if d.casefold() not in excluded and not d.startswith(".")
            ]
            for filename in filenames:
                if not filename.lower().endswith(tuple(sorted(extensoes))):
                    continue
                yield Path(dirpath) / filename
```
